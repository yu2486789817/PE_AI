-- MySQL 用户创建（替代 Oracle 版本）
-- 创建数据库
CREATE DATABASE IF NOT EXISTS se_project
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

-- 创建用户并授权
CREATE USER IF NOT EXISTS 'se'@'localhost' IDENTIFIED BY '2025SE';
CREATE USER IF NOT EXISTS 'se'@'%' IDENTIFIED BY '2025SE';

GRANT ALL PRIVILEGES ON se_project.* TO 'se'@'localhost';
GRANT ALL PRIVILEGES ON se_project.* TO 'se'@'%';

FLUSH PRIVILEGES;

-- 使用数据库
USE se_project;
